// Ver colección books
db.books.findOne()

// Listar categorías distintas
db.books.distinct("categories")

// Listar estados distintos
db.books.distinct("status")

// Libros cuyo longDescription inicia con letra A
db.books.find(
  { longDescription: { $gte: "A", $lt: "B" } },
  { title: 1, longDescription: 1 }
)

// Libros con longDescription entre A y B, exactamente 2 autores y publicados
db.books.find(
  { 
    longDescription: { $gte: "A", $lt: "B" },
    authors: { $size: 2 },
    status: "PUBLISH"
  },
  { 
    title: 1,
    longDescription: 1,
    status: 1,
    authors: 1
  }
)

// Convertir resultado a arreglo
db.books.find(
  { 
    longDescription: { $gte: "A", $lt: "B" },
    authors: { $size: 2 },
    status: "PUBLISH"
  },
  { 
    title: 1, 
    longDescription: 1, 
    status: 1, 
    authors: 1 
  }
).toArray()

// Recorrer resultados y mostrar formato personalizado
db.books.find(
  { 
    longDescription: { $gte: "A", $lt: "B" },
    authors: { $size: 2 },
    status: "PUBLISH"
  },
  { 
    title: 1, 
    longDescription: 1, 
    status: 1, 
    authors: 1 
  }
).forEach(function(valor, indice){
  print(
    "Titulo: " + valor.title + 
    " Author 1: " + valor.authors[0] + 
    " Author 2: " + valor.authors[1] + 
    " Registro No. " + indice
  );
});

// Tamaño de la colección en bytes
db.books.dataSize()

// Total de libros
db.books.countDocuments()

// Libros con 200 o más páginas
db.books.countDocuments({ pageCount: { $gte: 200 } })

// Libros entre 300 y 600 páginas (intervalo cerrado)
db.books.countDocuments({ 
  pageCount: { $gte: 300, $lte: 600 } 
})

// Libros con 0 páginas
db.books.countDocuments({ pageCount: 0 })

// Libros con páginas distintas de 0
db.books.countDocuments({ pageCount: { $ne: 0 } })

// Libros publicados
db.books.countDocuments({ status: "PUBLISH" })

// Libros no publicados
db.books.countDocuments({ status: { $ne: "PUBLISH" } })
